import { chromium } from '@playwright/test';
import { preview } from 'vite';
import { cpus, totalmem, platform, release } from 'node:os';
import { mkdir, writeFile } from 'node:fs/promises';
import { resolve } from 'node:path';

const server = await preview({ root: resolve('apps/web'), configFile: resolve('apps/web/vite.config.ts'), preview: { host: '127.0.0.1', port: 5177, strictPort: true, open: false } });
const browser = await chromium.launch({ channel: 'msedge', headless: true, chromiumSandbox: true });
const results: Record<string, unknown>[] = [];
const percentile = (values: number[], p: number) => [...values].sort((a, b) => a - b)[Math.min(values.length - 1, Math.ceil(values.length * p) - 1)];
const errors: string[] = [];
try {
  for (const count of [100, 300, 800]) for (const mode of ['collapsed', 'auto', 'expanded']) {
    const rounds: Record<string, number | string | null>[] = [];
    for (let round = 0; round < 3; round++) {
      const context = await browser.newContext({ viewport: { width: 1440, height: 1000 }, deviceScaleFactor: 1, reducedMotion: 'reduce' });
      const page = await context.newPage();
      page.on('pageerror', error => errors.push(error.message));
      await page.goto('http://127.0.0.1:5177/?count=' + count + '&mode=' + mode, { waitUntil: 'networkidle' });
      await page.waitForFunction(key => { const s = window.__skillTreePrototype?.snapshot(); return s?.readyKey === key && s.renderedCount > 0; }, count + ':' + mode);
      if (mode === 'auto') {
        await page.evaluate(() => { const el = document.querySelector<HTMLElement>('.graph-surface')!; return window.__skillTreePrototype!.viewport(el.clientWidth / 2, el.clientHeight / 2, 1.1); });
        await page.waitForFunction(() => { const s = window.__skillTreePrototype?.snapshot(); return Boolean(s?.layoutReady && s.zoom > 1 && s.visibleCount > 22); });
        await page.evaluate(() => new Promise<void>(resolveFrame => requestAnimationFrame(() => requestAnimationFrame(() => resolveFrame()))));
      }
      const initial = await page.evaluate(() => window.__skillTreePrototype!.snapshot());
      const frames = await page.evaluate(async () => {
        const api = window.__skillTreePrototype!;
        const origin = api.getViewport(), times: number[] = [];
        let last = performance.now(), index = 0;
        await new Promise<void>(resolveFrames => {
          function move(now: number) {
            if (index > 4) times.push(now - last);
            last = now;
            void api.viewport(origin.x + Math.sin(index / 14) * 24, origin.y + Math.cos(index / 17) * 18, origin.zoom * (1 + Math.sin(index / 20) * .035));
            if (++index < 100) requestAnimationFrame(move); else resolveFrames();
          }
          requestAnimationFrame(move);
        });
        await api.viewport(origin.x, origin.y, origin.zoom);
        return times;
      });
      const selectionMs = await page.evaluate(async () => {
        const button = [...document.querySelectorAll<HTMLButtonElement>('.suggested-role')].find(item => item.textContent?.includes('Frontend-разработчик'));
        if (!button) throw new Error('Кнопка выбора профессии не отрисована');
        const start = performance.now(); button.click();
        await new Promise<void>((resolveSelection, reject) => {
          const deadline = start + 5000;
          function check() {
            if (document.querySelector('[data-testid="selected-title"]')?.textContent === 'Frontend-разработчик') requestAnimationFrame(() => requestAnimationFrame(() => resolveSelection()));
            else if (performance.now() > deadline) reject(new Error('Карточка не обновилась')); else requestAnimationFrame(check);
          }
          requestAnimationFrame(check);
        });
        return performance.now() - start;
      });
      rounds.push({ round: round + 1, appReadyMs: initial.firstInteractionMs, catalogMs: initial.catalogMs, projectionMs: initial.projectionMs,
        canonicalCount: initial.canonicalCount, visibleCount: initial.visibleCount, renderedCount: initial.renderedCount, edgeCount: initial.edgeCount,
        fps: 1000 / (frames.reduce((a, b) => a + b, 0) / frames.length), frameP95Ms: percentile(frames, .95), selectionMs, heapBytes: initial.heapBytes });
      await context.close();
    }
    const entry = { count, mode, appReadyMedianMs: percentile(rounds.map(item => Number(item.appReadyMs)), .5),
      fpsMedian: percentile(rounds.map(item => Number(item.fps)), .5), frameP95WorstMs: Math.max(...rounds.map(item => Number(item.frameP95Ms))),
      selectionMedianMs: percentile(rounds.map(item => Number(item.selectionMs)), .5), rounds };
    results.push(entry); console.log(JSON.stringify({ count, mode, readyMs: Math.round(entry.appReadyMedianMs), fps: Math.round(entry.fpsMedian), selectionMs: Math.round(entry.selectionMedianMs) }));
  }
  const page = await browser.newPage({ viewport: { width: 1600, height: 1100 }, deviceScaleFactor: 1 });
  await page.goto('http://127.0.0.1:5177/');
  await page.waitForFunction(() => window.__skillTreePrototype?.snapshot().readyKey === '0:auto');
  await mkdir('docs/measurements', { recursive: true });
  await page.screenshot({ path: 'docs/measurements/prototype-overview.png', fullPage: true });
  await page.evaluate(() => window.__skillTreePrototype!.select('role:node'));
  await page.waitForFunction(() => document.querySelector('[data-testid="selected-title"]')?.textContent === 'Node.js-разработчик');
  await page.screenshot({ path: 'docs/measurements/prototype-node-route.png', fullPage: true });
  await writeFile('docs/measurements/graph-benchmark.json', JSON.stringify({
    measuredAtUtc: new Date().toISOString(), environment: { node: process.version, platform: platform(), osRelease: release(), cpu: cpus()[0]?.model,
      logicalProcessors: cpus().length, ramGiB: Math.round(totalmem() / 1024 ** 3), browser: await browser.version(), channel: 'msedge', headless: true,
      viewport: '1440x1000', deviceScaleFactor: 1, build: 'production', rounds: 3, frameSamplesPerRound: 95, note: 'RAF frames during programmatic pan/zoom; selection uses a DOM click and waits for card rendering. App-ready starts inside App, excluding download and browser startup. Heap is JS heap only.' }, results, errors
  }, null, 2));
  if (errors.length) throw new Error('Ошибки страницы: ' + errors.join('; '));
} finally {
  await browser.close();
  await new Promise<void>((resolveClose, reject) => server.httpServer.close(error => error ? reject(error) : resolveClose()));
}
